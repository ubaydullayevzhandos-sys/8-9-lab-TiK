codes = { "0": "A", "10": "B", "110": "C", "111": "D" }
message = "1011100110"
result = "" temp = ""
for bit in message: temp += bit
if temp in codes:
    result += codes[temp]
    temp = ""
print("\n2-есеп:") print("Бастапқы хабар:", message) print("Декодталған хабар:", result)
